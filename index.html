GENERACIÓN DE PDF LOGISVIAL CALIBRADO // ========================================== function generarPDFLogisvial(data) { if (data && data.estado === "ANULADO") { alert("⛔ OPERACIÓN DENEGADA: Este remito se encuentra ANULADO y no puede ser impreso."); return; } const { jsPDF } = window.jspdf; // Medidas de hoja continua: 340 mm x 210 mm const doc = new jsPDF({ orientation: 'landscape', unit: 'mm', format: [340, 210] }); const ox = parseFloat(document.getElementById("offset-x").value) || 0; const oy = parseFloat(document.getElementById("offset-y").value) || 0; // Desplazamiento general +3.5 mm a la derecha manteniendo el paso de 107.5 mm const columnasX = [ -6.5 + ox, // Cuerpo 1: Original 101.0 + ox, // Cuerpo 2: Duplicado 208.5 + ox // Cuerpo 3: Triplicado ]; // Alturas (Y) calibradas con la impresión const yFecha = 30.5 + oy; // Casillero FECHA (D / M / A) const yCliente = 44.0 + oy; // Renglón 1: Señor (es): const yDomLoc = 49.5 + oy; // Renglón 2: Domicilio (izq) | Localidad (der) // Módulo bajado para encajar sobre las líneas preimpresas: const yPatenteCuit = 58.5 + oy; // Renglón 3: Patente (izq) | CUIT Cliente (der) const yTransp = 64.0 + oy; // Renglón 4: Transportista (izq) | Datos fletero (der) const yItemsInicio = 76.0 + oy; // Renglón 1 de la tabla de materiales const saltoRenglon = 6.2; // Distancia entre renglones (hasta 9) // Observaciones bajadas al recuadro inferior const yObs = 168.0 + oy; // Línea dentro del recuadro de OBSERVACIONES // Desarmar la fecha (YYYY-MM-DD o DD-MM-YYYY) let dia = '', mes = '', anio = ''; if (data.fecha) { const partes = data.fecha.replace(/\//g, '-').split('-'); if (partes.length === 3) { if (partes[0].length === 4) { dia = partes[2]; mes = partes[1]; anio = partes[0].slice(-2); } else { dia = partes[0]; mes = partes[1]; anio = partes[2].slice(-2); } } else { dia = data.fecha; } } const inputObs = (data.obs || document.getElementById("remito-obs")?.value || "").trim().toUpperCase(); columnasX.forEach((colX) => { // 1. RENGLÓN 3: PATENTE (Izquierda) | CUIT CLIENTE (Derecha) if (data.transporte?.patente) { doc.text(data.transporte.patente.toUpperCase(), colX + 16, yPatenteCuit); } if (data.cliente?.cuit) { doc.text(String(data.cliente.cuit), colX + 61, yPatenteCuit); } // 5. RENGLÓN 4: TRANSPORTISTA (Izquierda) | DATOS FLETERO (Derecha) if (data.transporte?.transportista) { doc.text(data.transporte.transportista.toUpperCase().slice(0, 24), colX + 16, yTransp); } const refFletero = data.transporte?.cuit || data.transporte?.domicilio || ""; if (refFletero) { doc.text(String(refFletero).toUpperCase().slice(0, 16), colX + 61, yTransp); } // 6. DETALLE DE MATERIALES (Renglones 1 al 9) doc.setFont("courier", "bold"); doc.setFontSize(9); const itemsAImprimir = (data.items || []).slice(0, 9); itemsAImprimir.forEach((it, idx) => { const lineaY = yItemsInicio + (idx * saltoRenglon); if (it.cant) { doc.text(String(it.cant), colX + 10, lineaY, { align: "center" }); } const descTxt = (it.desc || '').toUpperCase(); doc.text(descTxt.slice(0, 38), colX + 22, lineaY); }); // 7. OBSERVACIONES (En su recuadro inferior) if (inputObs) { doc.setFont("courier", "bold"); doc.setFontSize(8); doc.text(inputObs.slice(0, 45), colX + 20, yObs); } }); // Abrir PDF para mandar a imprimir const blobUrl = URL.createObjectURL(doc.output('blob')); window.open(blobUrl, '_blank'); } async function anularRemito(idRemito) { const confirmacion = confirm("¿Estás seguro de que deseas ANULAR este remito?
