// ========================================== // 1. GENERACIÓN DE PDF LOGISVIAL CALIBRADO // ========================================== function generarPDFLogisvial(data) { // Bloqueo de seguridad: No generar PDF si el remito fue anulado if (data && data.estado === "ANULADO") { alert("⛔ OPERACIÓN DENEGADA: Este remito se encuentra ANULADO y no puede ser impreso."); return; } const { jsPDF } = window.jspdf; // Medidas exactas de papel continuo: 340 mm x 210 mm const doc = new jsPDF({ orientation: 'landscape', unit: 'mm', format: [340, 210] }); const ox = parseFloat(document.getElementById("offset-x")?.value) || 0; const oy = parseFloat(document.getElementById("offset-y")?.value) || 0; // Columnas horizontales calibradas para papel continuo (paso ~107.5 mm) const columnasX = [ -6.5 + ox, // Cuerpo 1: Original 101.0 + ox, // Cuerpo 2: Duplicado 208.5 + ox // Cuerpo 3: Triplicado ]; // Coordenadas verticales ajustadas al formulario preimpreso const yFecha = 30.5 + oy; // Casillero FECHA (D / M / A) const yCliente = 45.5 + oy; // Señor (es): const yDomLoc = 51.0 + oy; // Domicilio (izq) | Localidad (der) const yPatenteCuit = 59.5 + oy; // Patente (izq) | CUIT Cliente (der) const yTransp = 65.0 + oy; // Transportista (izq) | Datos fletero (der) const yItemsInicio = 77.0 + oy; // Primer renglón de materiales const saltoRenglon = 6.2; // Distancia entre renglones (hasta 9) const yObs = 186.0 + oy; // Recuadro inferior de OBSERVACIONES // Desarmar la fecha (YYYY-MM-DD o DD-MM-YYYY) let dia = '', mes = '', anio = ''; if (data.fecha) { const partes = data.fecha.replace(/\//g, '-').split('-'); if (partes.length === 3) { if (partes[0].length === 4) { dia = partes[2]; mes = partes[1]; anio = partes[0].slice(-2); } else { dia = partes[0]; mes = partes[1]; anio = partes[2].slice(-2); } } else { dia = data.fecha; } } const inputObs = (data.obs || document.getElementById("remito-obs")?.value || "").trim().toUpperCase(); columnasX.forEach((colX) => { // 1. FECHA (centrada en casilleros D | M | A) doc.setFont("courier", "bold"); doc.setFontSize(9); if (mes && anio) { doc.text(String(dia), colX + 73.0, yFecha, { align: "center" }); doc.text(String(mes), colX + 81.0, yFecha, { align: "center" }); doc.text(String(anio), colX + 89.0, yFecha, { align: "center" }); } else { doc.text(String(data.fecha || ''), colX + 80.0, yFecha, { align: "center" }); } // 2. SEÑOR (ES) doc.setFont("courier", "bold"); doc.setFontSize(8.5); const razon = (data.cliente?.razon || 'PARTICULAR').toUpperCase(); doc.text(razon.slice(0, 48), colX + 16, yCliente); // 3. DOMICILIO (Izq, max 24 car.) | LOCALIDAD (Der) doc.setFont("courier", "bold"); doc.setFontSize(8); if (data.cliente?.domicilio) { const domCorto = data.cliente.domicilio.toUpperCase().slice(0, 24); doc.text(domCorto, colX + 16, yDomLoc); } if (data.cliente?.localidad) { const locCorta = data.cliente.localidad.toUpperCase().slice(0, 16); doc.text(locCorta, colX + 61, yDomLoc); } // 4. PATENTE (Izq, +4 mm para despejar etiqueta) | CUIT CLIENTE (Der) if (data.transporte?.patente) { doc.text(data.transporte.patente.toUpperCase(), colX + 20, yPatenteCuit); } if (data.cliente?.cuit) { doc.text(String(data.cliente.cuit), colX + 61, yPatenteCuit); } // 5. TRANSPORTISTA / CHOFER (Izq) | DATOS FLETERO (Der) if (data.transporte?.transportista) { doc.text(data.transporte.transportista.toUpperCase().slice(0, 22), colX + 20, yTransp); } const refFletero = data.transporte?.cuit || data.transporte?.domicilio || ""; if (refFletero) { doc.text(String(refFletero).toUpperCase().slice(0, 16), colX + 61, yTransp); } // 6. MATERIALES (Renglones 1 al 9) doc.setFont("courier", "bold"); doc.setFontSize(9); const itemsAImprimir = (data.items || []).slice(0, 9); itemsAImprimir.forEach((it, idx) => { const lineaY = yItemsInicio + (idx * saltoRenglon); // Cantidad centrada en la casilla angosta if (it.cant) { doc.text(String(it.cant), colX + 8, lineaY, { align: "center" }); } // Descripción en columna ancha const descTxt = (it.desc || '').toUpperCase(); doc.text(descTxt.slice(0, 38), colX + 22, lineaY); }); // 7. OBSERVACIONES (En recuadro inferior) if (inputObs) { doc.setFont("courier", "bold"); doc.setFontSize(8); doc.text(inputObs.slice(0, 45), colX + 20, yObs); } }); // Abrir PDF para mandar a la Brother const blobUrl = URL.createObjectURL(doc.output('blob')); window.open(blobUrl, '_blank'); } // ========================================== // 2. FUNCIÓN PARA REIMPRIMIR CON VALIDACIÓN // ========================================== function reimprimirRemitoPDF(remito) { if (remito && remito.estado === "ANULADO") { alert("⛔ OPERACIÓN NO PERMITIDA: Este remito se encuentra ANULADO y no se puede imprimir."); return; } generarPDFLogisvial(remito); } // ========================================== // 3. FUNCIÓN PARA ANULAR SIN BORRAR // ========================================== async function anularRemito(idRemito) { const confirmacion = confirm("¿Estás seguro de que deseas ANULAR este remito? No se borrará del historial, pero quedará invalidado y bloqueado para impresión."); if (!confirmacion) return; try { await db.collection("remitos").doc(idRemito).update({ estado: "ANULADO", fechaAnulacion: new Date().toISOString() }); alert("Remito anulado con éxito."); if (typeof cargarHistorial === "function") { cargarHistorial(); } } catch (error) { console.error("Error al anular remito:", error); alert("Hubo un error al intentar anular el remito."); } } // ========================================== // 4. HISTORIAL CON BADGE Y BOTONES DE ESTADO // ========================================== function renderizarHistorial(lista) { const tbody = document.getElementById("historial-tbody"); const badge = document.getElementById("badge-count"); if (badge) badge.innerText = lista.length; if (lista.length === 0) { tbody.innerHTML = 'No se encontraron remitos en este período.'; return; } tbody.innerHTML = ""; lista.forEach(r => { const esAnulado = r.estado === "ANULADO"; const matResumen = r.items ? r.items.map(i => `${i.cant} ${i.unidad} - ${i.desc}`).join("
") : "-"; // Estilo atenuado si está anulado const estiloFila = esAnulado ? 'class="border-b border-rose-100 bg-rose-50/50 opacity-75"' : 'class="border-b border-slate-100 hover:bg-slate-50"'; // Número tachado + Badge rojo si está anulado const textoNumero = esAnulado ? `${r.numero || "-"} ANULADO` : `${r.numero || "-"}`; // Botón de reimpresión: si está anulado queda deshabilitado en gris const botonReimprimir = esAnulado ? `` : ``; // Botón de anular: si ya está anulado muestra la marca fija; si está activo permite anular const botonAnular = esAnulado ? `❌` : ``; tbody.innerHTML += ` ${r.fecha} ${textoNumero} ${r.cliente?.razon || "-"} ${r.transporte?.patente || "-"} ${matResumen}
${botonReimprimir} ${botonAnular}
`; }); }
